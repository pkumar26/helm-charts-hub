  # {{ if .Values.gatewayApi.gateway.name }}2{{ else }}3{{ end }}. Create an HTTPRoute to send traffic to the app
  kubectl apply -f - <<EOF
  apiVersion: gateway.networking.k8s.io/v1
  kind: HTTPRoute
  metadata:
    name: httpbin-route
  spec:
    parentRefs:
      - name: {{ .Values.gatewayApi.gateway.name | default "traefik-gateway" }}
    rules:
      - matches:
          - path:
              type: PathPrefix
              value: /
        backendRefs:
          - name: httpbin
            port: 80
  EOF

  # {{ if .Values.gatewayApi.gateway.name }}3{{ else }}4{{ end }}. Verify the Gateway is programmed
  kubectl get gateway {{ .Values.gatewayApi.gateway.name | default "traefik-gateway" }}

  # {{ if .Values.gatewayApi.gateway.name }}4{{ else }}5{{ end }}. Test the route

Next Steps (Ingress mode) — deploy a sample app and route traffic:

  # 1. Deploy a sample application
  kubectl create deployment httpbin --image=kennethreitz/httpbin --port=80
  kubectl expose deployment httpbin --port=80

  # 2. Create an Ingress resource to route traffic to the app
  kubectl apply -f - <<EOF
  apiVersion: networking.k8s.io/v1
  kind: Ingress
  metadata:
    name: httpbin-ingress
  spec:
    ingressClassName: {{ .Values.ingressController.ingressClassName }}
    rules:
      - http:
          paths:
            - path: /
              pathType: Prefix
              backend:
                service:
                  name: httpbin
                  port:
                    number: 80
  EOF

  # 3. Verify the Ingress is created and has an address
  kubectl get ingress httpbin-ingress

  # 4. Test the route
  TRAEFIK_IP=$(kubectl get svc {{ include "common-lib.fullname" . }} -n {{ .Release.Namespace }} -o jsonpath='{.status.loadBalancer.ingress[0].ip}')
  curl -s http://$TRAEFIK_IP/get | head -20

{{- end }}
