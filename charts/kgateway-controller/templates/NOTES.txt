Next Steps — create a Gateway, deploy a sample app, and route traffic:

  # 1. Create a Gateway (this triggers kgateway to provision Envoy proxy pods)
  kubectl apply -f - <<EOF
  apiVersion: gateway.networking.k8s.io/v1
  kind: Gateway
  metadata:
    name: my-gateway
  spec:
    gatewayClassName: {{ .Values.gatewayApi.gatewayClassName }}
    listeners:
      - name: http
        protocol: HTTP
        port: 8080
  EOF

  # 2. Deploy a sample application
  kubectl create deployment httpbin --image=kennethreitz/httpbin --port=80
  kubectl expose deployment httpbin --port=80

  # 3. Create an HTTPRoute to send traffic to the app
  kubectl apply -f - <<EOF
  apiVersion: gateway.networking.k8s.io/v1
  kind: HTTPRoute
  metadata:
    name: httpbin-route
  spec:
    parentRefs:
      - name: my-gateway
    rules:
      - matches:
          - path:
              type: PathPrefix
              value: /
        backendRefs:
          - name: httpbin
            port: 80
  EOF

  # 4. Verify the Gateway is accepted and proxy pods are running
  kubectl get gateway my-gateway
  kubectl get pods -l gateway.networking.k8s.io/gateway-name=my-gateway

  # 5. Test the route (port-forward if using ClusterIP)
  kubectl port-forward svc/my-gateway-http-8080 8080:8080 &
  curl -s http://localhost:8080/get | head -20
